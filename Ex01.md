## AIM

To write and execute a C program to implement a symbol table that stores identifiers along with their corresponding memory addresses and allows searching for a particular identifier.

## ALGORITHM

1. Start the program.
2. Get the input expression from the user, terminating it with the symbol `$`.
3. Read the input character by character.
4. Check whether the character represents a variable (identifier).
5. If the next character is an operator or a special symbol, allocate memory dynamically for the variable using the `malloc()` function.
6. Insert the variable into the symbol table along with its corresponding memory address.
7. Repeat the above steps until the terminating symbol `$` is encountered.
8. To search for a variable, get the variable name from the user.
9. Search the symbol table for the corresponding variable.
10. If the variable is found, display the variable along with its memory address.
11. Stop the program.

## PROGRAM

```c

#include <stdio.h>
#include <stdlib.h>
#include <ctype.h>
#include <string.h>

struct symbol
{
    char name[20];
    void *address;
};

int main()
{
    struct symbol table[50];
    int count = 0, i, found;
    char input[100], search[20], temp[20];
    int j;

    printf("Enter the input expression (end with $): ");
    scanf(" %99[^\n]", input);

    for (i = 0; input[i] != '$' && input[i] != '\0'; i++)
    {
        if (isalpha((unsigned char)input[i]))
        {
            j = 0;

            while (isalnum((unsigned char)input[i]) && j < 19)
            {
                temp[j++] = input[i++];
            }

            temp[j] = '\0';
            i--;

            found = 0;

            for (int k = 0; k < count; k++)
            {
                if (strcmp(table[k].name, temp) == 0)
                {
                    found = 1;
                    break;
                }
            }

            if (!found && count < 50)
            {
                strcpy(table[count].name, temp);
                table[count].address = malloc(1);

                if (table[count].address == NULL)
                {
                    printf("Memory allocation failed!\n");
                    return 1;
                }

                count++;
            }
        }
    }

    printf("\nSYMBOL TABLE\n");
    printf("-------------------------\n");
    printf("VARIABLE\tADDRESS\n");
    printf("-------------------------\n");

    for (i = 0; i < count; i++)
    {
        printf("%s\t\t%p\n", table[i].name,
               table[i].address);
    }

    printf("\nEnter the variable to be searched: ");
    scanf("%19s", search);

    found = 0;

    for (i = 0; i < count; i++)
    {
        if (strcmp(table[i].name, search) == 0)
        {
            printf("Variable found!\n");
            printf("Variable: %s\n", table[i].name);
            printf("Address: %p\n", table[i].address);
            found = 1;
            break;
        }
    }

    if (!found)
    {
        printf("Variable not found in the symbol table.\n");
    }

    for (i = 0; i < count; i++)
    {
        free(table[i].address);
    }

    return 0;
}

```
